# Alert Dispatch Reliability Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make every criteria-meeting portfolio alert reliably reach email + Telegram, stop the re-email storm on backend restart, and keep an inspectable, auto-pruned record of every message actually sent.

**Architecture:** Four independent changes. (1) Remove the browser-side "conflict" skip so held positions always alert. (2) Persist the alert-notifier dedup state to a JSON file and switch its time base from `monotonic()` to wall-clock so a restart doesn't re-send. (3) Record every successfully sent email/Telegram to a dedicated JSONL log with a configurable retention pruner. (4) Expose a retention-days field in the alert panel (reminder-interval field already exists — verify only).

**Tech Stack:** FastAPI + Pydantic v2 + pytest (backend, `backend/`), Next.js 15 + TypeScript + Jest + Zustand (frontend, `frontend/`). Backend run from `backend/`: `pytest`. Frontend run from `frontend/`: `npm run test`, `npm run typecheck`.

# Persisted across restarts so a backend restart does NOT re-email every
# currently-breached position. Stored as wall-clock seconds (time.time()),
# NOT monotonic — monotonic resets each process and is meaningless on disk.
_DEDUP_PATH = Path(__file__).parent.parent.parent.parent / 'alert_dedup.json'


def _save_dedup() -> None:
    """Persist dedup state. Caller must hold _fired_lock."""
    try:
        with open(_DEDUP_PATH, 'w', encoding='utf-8') as f:
            json.dump({'fired': _fired, 'first_seen': _first_seen}, f)
    except Exception:
        logger.debug('Could not persist dedup state', exc_info=True)


def _load_dedup() -> None:
    """Load dedup state from disk into the in-memory dicts (idempotent)."""
    try:
        if _DEDUP_PATH.exists():
            with open(_DEDUP_PATH, encoding='utf-8') as f:
                data = json.load(f) or {}
            _fired.update({k: float(v) for k, v in data.get('fired', {}).items()})
            _first_seen.update(data.get('first_seen', {}))
    except Exception:
        logger.debug('Could not load dedup state', exc_info=True)

```python
"""Persistent JSONL record of email/Telegram messages actually sent, with
date-based retention. Kept SEPARATE from audit.log so retention pruning never
touches scan/approval history."""
from __future__ import annotations

import json
import logging
from datetime import datetime, timedelta, timezone
from pathlib import Path

logger = logging.getLogger(__name__)

# backend/app/services/notifications/ -> 4 parents = backend/
_SENT_LOG_PATH = Path(__file__).parent.parent.parent.parent / 'sent_messages.log'


def log_sent(channel: str, recipient: str, subject: str) -> None:
    """Append one record for a message that was actually delivered."""
    entry = {
        'ts': datetime.now(tz=timezone.utc).isoformat(),
        'channel': channel,
        'recipient': recipient,
        'subject': subject,
    }
    try:
        with open(_SENT_LOG_PATH, 'a', encoding='utf-8') as f:
            f.write(json.dumps(entry, ensure_ascii=False) + '\n')
    except Exception:
        logger.debug('Could not write sent-message log', exc_info=True)


def prune_sent_log(retention_days: int) -> None:
    """Drop records older than retention_days. Unparseable lines are kept."""
    if retention_days <= 0 or not _SENT_LOG_PATH.exists():
        return
    cutoff = datetime.now(tz=timezone.utc) - timedelta(days=retention_days)
    try:
        with open(_SENT_LOG_PATH, encoding='utf-8') as f:
            lines = f.readlines()
        kept: list[str] = []
        for line in lines:
            try:
                ts = datetime.fromisoformat(json.loads(line)['ts'])
                if ts >= cutoff:
                    kept.append(line)
            except Exception:
                kept.append(line)
        with open(_SENT_LOG_PATH, 'w', encoding='utf-8') as f:
            f.writelines(kept)
    except Exception:
        logger.debug('Could not prune sent-message log', exc_info=True)
```
