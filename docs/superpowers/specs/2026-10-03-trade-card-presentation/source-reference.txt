function recomputeTargetEstOption(
  targetStockPrice: number,
  entryLimit: number,
  data: TradeCardData,
): number {
  const cd = data.contractDetails;
  if (!cd) return 0;
  const isCall = cd.right === 'C';
  const strike = cd.strike;
  const currentStockPrice = data.price;
  const delta =
    (isCall && targetStockPrice > strike) || (!isCall && targetStockPrice < strike)
      ? 0.5
      : 0.3;
  const priceMove = Math.abs(targetStockPrice - currentStockPrice);
  return Math.max(0.01, entryLimit + delta * priceMove);
}

function rrContribution(
  estOptionPrice: number,
  entryLimit: number,
  riskBasis: number,
): number {











  if (riskBasis <= 0 || entryLimit <= 0) return 0;
  const reward = estOptionPrice - entryLimit;
  return Math.round((reward / riskBasis) * 10) / 10;
}

function normalizeStructuralStopResponse(res: {
  available: boolean;
  price: number | null;
  source: 'order_block' | 'swing_low' | 'swing_high' | null;
  confidence: number | null;
  distancePct: number | null;
  stopCost?: StopCostOut | null;
}): StructuralStop | null {
  return res.available && res.price != null && res.source != null
    && res.confidence != null && res.distancePct != null







    ? { price: res.price, source: res.source, confidence: res.confidence,
        distancePct: res.distancePct, stopCost: res.stopCost ?? null }
    : null;
}




function formatHHMM(iso: string): string {
  const d = new Date(iso);
  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
}

function structuralStopSourceLabel(source: 'order_block' | 'swing_low' | 'swing_high' | null | undefined): string {
  if (source === 'order_block') return 'Structural stop — Order Block';
  if (source === 'swing_low') return 'Structural stop — Swing Low';
  if (source === 'swing_high') return 'Structural stop — Swing High';
  return 'Structural stop';
}

function generateRecId(): string {
  return 'REC-' + Date.now() + '-' + Math.random().toString(36).slice(2, 6).toUpperCase();
}


const REQUEST_TIMEOUT_MS = 20_000;

interface RequestDeadline {
  timer: ReturnType<typeof setTimeout>;
  
  timedOut: boolean;
}

function abortAfterTimeout(controller: AbortController): RequestDeadline {
  const deadline: Partial<RequestDeadline> = { timedOut: false };
  deadline.timer = setTimeout(() => {
    deadline.timedOut = true;
    controller.abort();
  }, REQUEST_TIMEOUT_MS);
  return deadline as RequestDeadline;
}


function isFinished(controller: AbortController): boolean {
  return controller.signal.aborted;
}


function isRealFailure(controller: AbortController, deadline: RequestDeadline): boolean {
  return !controller.signal.aborted || deadline.timedOut;
}

function EditableTargetRow({
  target,
  index,
  editedStockPrice,
  entryLimit,
  premiumStop,
  data,
  onStockPriceChange,
  totalContracts,
}: {
  target: TargetLevel;
  index: number;
  editedStockPrice: number;
  entryLimit: number;
  premiumStop: number;
  data: TradeCardData;
  onStockPriceChange: (v: number) => void;
  totalContracts: number;
}) {
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(String(editedStockPrice ?? 0));

  const startEdit = useCallback(() => {
    setDraft(String(editedStockPrice ?? 0));
    setEditing(true);
  }, [editedStockPrice]);

  const save = useCallback(() => {
    const parsed = parseFloat(draft);
    if (!isNaN(parsed) && parsed > 0) {
      onStockPriceChange(Math.round(parsed * 100) / 100);
    }
    setEditing(false);
  }, [draft, onStockPriceChange]);

  const handleKey = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === 'Enter') save();
      if (e.key === 'Escape') setEditing(false);
    },
    [save],
  );

  const originalStockPrice = target.targetPrice ?? target.stockPrice;
  const isTargetEdited = editedStockPrice !== originalStockPrice;

  const estOptionPrice = isTargetEdited
    ? recomputeTargetEstOption(editedStockPrice, entryLimit, data)
    : (target.estimatedOptionPrice ?? 0);

  const rr = rrContribution(estOptionPrice, entryLimit, premiumStop);

  const statusVal = target.status;
  const rowTone = statusVal === 'HIT' ? 'pass' : statusVal === 'MISSED' ? 'reject' : undefined;
  const statusClass = statusVal ? styles[`target_${statusVal}`] : styles.target_PENDING;
  const badgeStatus = statusVal === 'HIT' ? 'PASS' : statusVal === 'MISSED' ? 'REJECT' : 'PENDING';
  const isScaleOut = target.contractsToSell !== undefined && target.contractsToSell < totalContracts;
  const label = target.id ? target.id.toUpperCase() : `T${index + 1}`;

  return (
    <div className={`${styles.targetRow} ${statusClass}`}>
      <div className={styles.targetMain}>
        <div className={styles.targetId}>
          <Text size={12} weight={600} tone={rowTone}>{label}</Text>
          {statusVal && <StatusBadge status={badgeStatus as 'PENDING' | 'PASS' | 'REJECT'} tooltip={targetStatusTooltip(statusVal)} />}
          {isTargetEdited && (
            <span className={styles.editBadge}>
              <Text size={10} tone="accent">edited</Text>
            </span>
          )}
        </div>
        <div className={styles.targetValues}>
          <div className={styles.kvRow}>
            <Text size={12} tone="secondary">Stock price</Text>
            {editing ? (
              <input
                type="number"
                className={styles.editTargetInput}
                value={draft}
                min={0.01}
                onChange={(e) => setDraft(e.target.value)}
                onBlur={save}
                onKeyDown={handleKey}
                autoFocus
                aria-label={`Edit target stock price for ${label}`}
              />
            ) : (
              <span
                className={styles.editValue}
                onClick={startEdit}
                role="button"
                tabIndex={0}
                onKeyDown={(e) => {
                  if (e.key === 'Enter' || e.key === ' ') startEdit();
                }}
                aria-label={`Edit target stock price for ${label}`}
              >
                <Data size={13} weight={500} tone={rowTone}>
                  ${(editedStockPrice ?? 0).toFixed(2)}
                </Data>
              </span>
            )}
          </div>
          {estOptionPrice > 0 && (
            <KV label="Est. option" value={`$${estOptionPrice.toFixed(2)}`} mono tone={rowTone} />
          )}
          {estOptionPrice > 0 && entryLimit > 0 && premiumStop > 0 && (
            <KV label="R:R" value={`${rr.toFixed(1)}x`} mono tone={rr < 0 ? 'reject' : undefined} />
          )}
          {target.contractsToSell !== undefined && (
            <KV label="Contracts" value={`${target.contractsToSell}`} mono />
          )}
        </div>
        {isScaleOut && target.contractsToSell !== undefined && (
          <Text size={11} tone="caution">
            Scale out: sell {target.contractsToSell} of {totalContracts}
          </Text>
        )}
      </div>
      {target.pricingMethod && (
        <span className={styles.pricingBadge}>
          <Text size={10} tone="dim">{target.pricingMethod}</Text>
        </span>
      )}
    </div>
  );
}


export function EditableKV({
  label,
  value,
  onChange,
  onRevert,
  min,
  max,
  mono,
  tone,
  prefix = '',
  suffix = '',
  decimals = 2,
  nullLabel = 'not computed',
}: {
  label: string;
  
  value: number | null;
  onChange: (v: number) => void;
  
  onRevert?: () => void;
  min?: number;
  max?: number;
  mono?: boolean;
  tone?: 'pass' | 'caution' | 'reject';
  prefix?: string;
  suffix?: string;
  decimals?: number;
  
  nullLabel?: string;
}) {
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(value == null ? '' : String(value));







  const [rejected, setRejected] = useState<string | null>(null);

  
  const typedRef = useRef(false);

  const startEdit = useCallback(() => {
    setDraft(value == null ? '' : String(value));
    setRejected(null);
    typedRef.current = false;
    setEditing(true);
  }, [value]);

  const save = useCallback(() => {
    if (!typedRef.current) {
      setEditing(false);
      return;
    }


















    const trimmed = draft.trim();
    const isCompleteNumber = /^-?(?:\d+(?:\.\d+)?|\.\d+)$/.test(trimmed);
    if (trimmed !== '' && !isCompleteNumber) {





      setRejected(`That is not a complete number — kept the previous value.`);
      setEditing(false);
      return;
    }
    const parsed = parseFloat(draft);
    if (isNaN(parsed)) {



      if (onRevert && draft.trim() === '') onRevert();
      setEditing(false);
      return;
    }
    const minVal = min ?? -Infinity;
    const maxVal = max ?? Infinity;



































    if (parsed === value) {
      setEditing(false);
      return;
    }
    if (parsed < minVal || parsed > maxVal) {
      setRejected(
        maxVal === Infinity
          ? `Must be at least ${prefix}${minVal.toFixed(decimals)}${suffix} — kept the previous value.`
          : `Must be between ${prefix}${minVal.toFixed(decimals)}${suffix} and `
            + `${prefix}${maxVal.toFixed(decimals)}${suffix} — kept the previous value.`,
      );
      setEditing(false);
      return;
    }

    const scale = 10 ** decimals;
    const next = Math.round(parsed * scale) / scale;



























    if (next === value) {
      setEditing(false);
      return;
    }
    setRejected(null);
    onChange(next);
    setEditing(false);
  }, [draft, min, max, onChange, onRevert, decimals, value, prefix, suffix]);

  const handleKey = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === 'Enter') save();
      if (e.key === 'Escape') setEditing(false);
    },
    [save],
  );

  const displayValue = value == null
    ? nullLabel
    : decimals === 0
      ? `${prefix}${value}${suffix}`
      : `${prefix}${value.toFixed(decimals)}${suffix}`;

  return (
    <div className={`${styles.kvRow} ${styles.kvRowEditable}`}>
      <Text size={12} tone="secondary">{label}</Text>
      {editing ? (
        <input

















          type="text"
          inputMode="decimal"
          className={styles.editInput}
          value={draft}
          onChange={(e) => { typedRef.current = true; setDraft(e.target.value); }}
          onBlur={save}
          onKeyDown={handleKey}
          autoFocus
          aria-label={`Edit ${label}`}





          title={onRevert ? `Clear to go back to the plan's ${label.toLowerCase()}` : undefined}
        />
      ) : (
        <span
          className={styles.editValue}
          onClick={startEdit}
          role="button"
          tabIndex={0}
          onKeyDown={(e) => {
            if (e.key === 'Enter' || e.key === ' ') startEdit();
          }}
          aria-label={`Edit ${label}`}
        >
          {value == null
            ? <Text size={13} weight={400} tone="dim">{displayValue}</Text>
            : mono
              ? <Data size={13} weight={500} tone={tone}>{displayValue}</Data>
              : <Text size={13} weight={500}>{displayValue}</Text>}
        </span>
      )}
      {}
      <span
        className={rejected ? styles.editRejected : styles.editRejectedEmpty}
        aria-live="polite"
      >
        {rejected ? <Text size={11} tone="caution">{rejected}</Text> : ''}
      </span>
    </div>
  );
}

function MaxLossOverrideEditor({
  override,
  limitDollar,
  onApply,
}: {
  override: MaxLossOverride | null;
  limitDollar: number;
  onApply: (v: MaxLossOverride | null) => void;
}) {
  const [editing, setEditing] = useState(false);
  const [draftMode, setDraftMode] = useState<'dollar' | 'percent'>(override?.mode ?? 'percent');
  const [draftValue, setDraftValue] = useState(() =>
    override
      ? (override.mode === 'percent' ? override.value * 100 : override.value)
      : 0,
  );

  const startEdit = () => {
    setDraftMode(override?.mode ?? 'percent');
    setDraftValue(override ? (override.mode === 'percent' ? override.value * 100 : override.value) : 0);

    const value = draftMode === 'percent' ? draftValue / 100 : draftValue;
    onApply({ mode: draftMode, value });

    onApply(null);
    setEditing(false);
  };

  return (
    <div className={styles.structuralStop}>
      <Text size={10} tone="dim">Max loss budget</Text>
      <Text size={11} tone="secondary">
        ${Math.round(limitDollar).toLocaleString('en-US')}

          : ' (account default)'}
      </Text>
      {!editing ? (
        <button type="button" className={styles.btn} onClick={startEdit}>
          <Text size={12} weight={500}>{override ? 'Edit override' : 'Set override'}</Text>

function rowOfPick(rows: MethodPickRow[], pick: MethodPick | null): string | null {
  if (!pick) return null;

  const hit = rows.find(
    (r) => Object.is(r.rowStockPrice, pick.rowStockPrice)
        && Object.is(r.rowOptionPrice, pick.rowOptionPrice),
  );
  return hit?.key ?? null;
}

}

  async function placeDirectOrder(orderType: 'LMT' | 'MKT') {
    const card = effectiveCard;
    if (!card || !selectedBroker) return;

    if (card.earningsVeto === true) return;




    if (!checkInvalidation(invalidationFor(card))) return;

    if (!checkMethodPick(methodPickFor(card))) return;


    setOrderError(null); setSaveNotes([]); setBasketNotes([]); setBasketError(null);
    const broker = selectedBroker;
    const blocked = directOrderBlock(orderType, broker);
    if (blocked) { setOrderError(blocked); return; }
    const setPlacing = orderType === 'LMT' ? setLimitPlacing : setMarketPlacing;
    setPlacing(true);
    setPaperOrderId(null);
    let label = card.contract;
    try {


      label = contractLabelFor(card);
      const brokerLabel = brokerLabelFor(broker);


      const contracts = Math.max(1, orderContractsBySymbol[card.symbol] ?? 1);


      const optionSymbol = toOptionSymbol(card);
      const limitPrice = orderType === 'LMT' ? entryLimitFor(card) : null;
      const pos = buildPaperPosition(card, orderType, broker, 'pending', liveQuoteForCard?.price, contracts);
      const recId = pos.recId;


      if (usePortfolioStore.getState().positions.some((p) => p.contract === pos.contract)) {
        setOrderError(`${label}: No order was sent — a position for this contract is already tracked in Portfolio. Close or edit that one instead.`);
        return;
      }

      const saved = await persistPosition(pos);






      const bare5xx = !saved.ok && !saved.offline && /^HTTP 5\d\d$/.test(saved.detail ?? '');
      const unparsed2xx = saved.ok && saved.entryPrice == null;
      if (saved.offline || bare5xx || unparsed2xx) {
        setOrderError(unparsed2xx
          ? `${label}: Portfolio saved it but its reply could not be read, so NO order was sent.`
          : `${label}: Portfolio tracking could not be confirmed, so NO order was sent.`);
        addDirectNote({ recId, kind: 'save-unconfirmed', label, brokerLabel, orderId: null });
        dropLocal(recId);
        try { await removeTrackedRow(recId, false); } catch {  }
        return;
      }
      if (!saved.ok) {



        const detail = saved.detail ?? 'not saved';
        setOrderError(/already exists/.test(detail)
          ? `${label}: No order was sent — a position for this contract is already tracked in Portfolio. Close or edit that one instead.`
          : `${label}: No order was sent — ${detail.replace(/\.$/, '')}.`);
        return;
      }



      const orderQty = saved.adjustedQty ?? contracts;
      const keepRow = (why: string, orderId: string | null) => {

        addDirectNote({ recId, kind: 'order-unknown', label, brokerLabel, orderId });
        const banner = `${label}: Order status UNKNOWN (${why}).`;
        setOrderError(banner);
        try {
          applyPersistedPosition(
            saved, { ...pos, contracts: orderQty }, addPosition,
            (q) => addSaveNote(`${label}: Portfolio sized it to ${q}; the order was requested for ${q}, status unknown.`),
            (r, why, kept) => addSaveNote(`${label}: ${droppedStopNote(r, why, kept)}`),
          );
        } catch (e) {
          setOrderError(`${banner} The local view failed to update: ${String(e)}`);
        }
      };
      try {
        let outcome: DirectOutcome;
        try {
          const res = await api.postOptionOrder({
            broker,
            symbol:        card.symbol,
            option_symbol: optionSymbol,
            action:        'BUY',
            effect:        'OPEN',
            quantity:      orderQty,
            order_type:    orderType,
            limit_price:   limitPrice,
            tif:           'DAY',
          });
          outcome = classifyDirectFailure({ response: res });
        } catch (e) {
          outcome = classifyDirectFailure({ thrown: e });
        }

        if (outcome.verdict === 'delete') {

          try {
            await removeTrackedRow(recId, true);
            setOrderError(`${label}: Order NOT placed: ${outcome.why}. The tracked position was removed.`);
          } catch {
            setOrderError(`${label}: Order NOT placed: ${outcome.why}.`);
            addDirectNote({ recId, kind: 'not-removed', label, brokerLabel, orderId: null });
          }
          return;
        }
        if (outcome.verdict === 'keep') {
          keepRow(outcome.why, outcome.orderId);
          return;
        }

        const orderId = outcome.orderId;
        try {
          applyPersistedPosition(
            saved,
            { ...pos, contracts: orderQty, sourceLabel: `${brokerLabel} — ${orderType === 'LMT' ? 'Limit' : 'Market'} (ID: ${orderId})` },
            addPosition,
            (q) => addSaveNote(`${label}: ${sizeNote(q)}`),
            (r, why, kept) => addSaveNote(`${label}: ${droppedStopNote(r, why, kept)}`),
          );
        } catch (e) {
          setOrderError(`${label}: Order placed (ID ${orderId}) but the local view failed to update: ${String(e)}`);
        }
        setPaperOrderId({ label, id: orderId });
      } catch (e) {


        keepRow(String(e), null);
      }
    } catch (e) {

      setOrderError(`${label}: No order was sent — ${String(e)}`);
    } finally {
      setPlacing(false);
    }
  }

interface Gate {
  label:  string;
  status: 'PASS' | 'CAUTION' | 'REJECT';
  
  detail: string;
}

function buildChecklist(
  candidate: TradeCardData,
  wlRow: WatchlistRow | undefined,
  regime: Status,
  horizonKey: ThresholdHorizon,
  ivThresholds: Record<ThresholdHorizon, HorizonIvThresholds> | null,
  rrPending: boolean,
  ivOverride: { value: number; yaml: number } | null = null,
  eventWindow: { reject_days: number; caution_days: number } | undefined = undefined,
): Gate[] {
  const gates: Gate[] = [];

  if (regime !== 'PENDING' && regime !== 'NO_DATA') {
    const regimeStatus: 'PASS' | 'CAUTION' | 'REJECT' =
      regime === 'PASS' ? 'PASS' : regime === 'CAUTION' ? 'CAUTION' : 'REJECT';
    gates.push({ label: 'Regime PASS', status: regimeStatus, detail: `Current market regime: ${regime}.` });
  }





  const iv = resolveIv(candidate.contractDetails);







  const gateDisabled = ivThresholds !== null
    && getIvPassMax(horizonKey, ivThresholds, iv.basis === 'none' ? 'absolute' : iv.basis) === null;
  if (ivThresholds === null) {
    gates.push({
      label: 'IV acceptable',
      status: 'CAUTION',
      detail: 'IV thresholds could not be loaded from scoring_rules.yaml — the IV gate was not applied.',
    });
  } else if (gateDisabled) {
    gates.push({
      label: 'IV acceptable',
      status: 'PASS',
      detail: `No IV gate for ${horizonKey}.`,
    });
  } else if (iv.basis === 'none') {
    gates.push({
      label: 'IV acceptable',
      status: 'CAUTION',
      detail: 'No implied-volatility data for this contract — the IV gate could not be applied.',
    });
  } else {
    const ivValue      = iv.value as number;
    const shown        = formatIv(iv, candidate.symbol);
    const ivPassMax    = getIvPassMax(horizonKey, ivThresholds, iv.basis);
    const ivCautionMax = getIvCautionMax(horizonKey, ivThresholds, iv.basis);
    const noun         = iv.basis === 'rank' ? 'IV rank' : 'IV';
    const ivStatus: 'PASS' | 'CAUTION' | 'REJECT' =
      ivPassMax === null    ? 'PASS' :           // Intraday: no IV gate
      ivValue <= ivPassMax  ? 'PASS' :
      ivCautionMax !== null && ivValue <= ivCautionMax ? 'CAUTION' :
      'REJECT';
    gates.push({
      label: 'IV acceptable',
      status: ivStatus,
      detail: ivPassMax === null
        ? `${noun} ${shown.text} — no IV gate for ${horizonKey}.`
        : `${noun} ${shown.text} vs. pass ≤${ivPassMax}${ivCautionMax != null ? ` / caution ≤${ivCautionMax}` : ''}.`

          + (ivOverride !== null && iv.basis === 'absolute'
            ? ` (session override from Settings; scoring_rules.yaml: ${ivOverride.yaml})` : ''),
    });
  }








  if (candidate.earningsUnknown) {
    gates.push(candidate.isEtf
      ? {
          label: 'Earnings date',
          status: 'PASS',
          detail: 'ETF — no earnings date to check.',
        }
      : {
          label: 'Earnings date',
          status: 'CAUTION',
          detail: 'No earnings date on file — the card is capped at CAUTION and auto-trade will not fire a single stock. Check the calendar before sending.',
        });
  }




  const days = candidate.earningsDays;
  if (days != null) {
    gates.push(eventWindow
      ? {
          label: 'Earnings outside reject window',
          status: days < eventWindow.reject_days ? 'REJECT' : days < eventWindow.caution_days ? 'CAUTION' : 'PASS',
          detail: `Next event in ${days}d — reject <${eventWindow.reject_days}d, caution <${eventWindow.caution_days}d (this scan's window).`,
        }
      : {
          label: 'Earnings outside reject window',
          status: 'CAUTION',
          detail: `Next event in ${days}d — this scan carried no event window; re-run the scan.`,
        });
  }

  const minScore = Math.min(candidate.stockScore, candidate.contractScore);
  gates.push({
    label: 'Score ≥ 55',
    status: statusForScore(minScore) as 'PASS' | 'CAUTION' | 'REJECT',
    detail: `Lower of stock score (${candidate.stockScore}) and contract score (${candidate.contractScore}) = ${minScore}.`,
  });


  if (rrPending) {
    gates.push({
      label: 'R:R ≥ 2.0',
      status: 'CAUTION',
      detail: 'R:R not yet recomputed for the contract picked in Step 6.',
    });
  } else gates.push({
    label: 'R:R ≥ 2.0',
    status: candidate.rrRatio >= 2.0 ? 'PASS' : candidate.rrRatio >= 1.5 ? 'CAUTION' : 'REJECT',
    detail: `Projected reward:risk is ${candidate.rrRatio.toFixed(1)}× — pass ≥2.0×, caution 1.5-2.0×.`,
  });

  const spreadScore = candidate.contractBreakdown.spread;
  gates.push({
    label: 'Spread < 5% of price',
    status: spreadScore >= 70 ? 'PASS' : spreadScore >= 40 ? 'CAUTION' : 'REJECT',
    detail: `Spread quality sub-score ${spreadScore}/100 — pass ≥70, caution 40-69.`,
  });

  return gates;
}

      {autoMode === 'post_scan' && (
        <Text size={11} tone="secondary">
          The system will automatically place this trade when the next scan completes with a PASS candidate.
          Run or schedule a scan to activate.
        </Text>
      )}

      {lastOrderId && (lastFireBooked === false


        ? <Text size={12} tone="caution">Order sent — ID: {lastOrderId} (not booked)</Text>
        : <Text size={12} tone="pass">Auto-trade fired — Order ID: {lastOrderId}</Text>
      )}
      {error && <Text size={12} tone="reject">{error}</Text>}
      {fireNote && <Text size={12} tone="caution">{fireNote}</Text>}

      {dupWarning && (
        <ConfirmDialog
          title={`You already hold ${optionSymbol}`}
          body={
            `Existing position: ${dupWarning.contracts} contract${dupWarning.contracts === 1 ? '' : 's'} `
            + `at $${dupWarning.entryPrice.toFixed(2)}. Arming will place a second buy-to-open on this contract `
            + 'when it fires. That order will NOT be added to your portfolio (a contract is booked once) — '
            + "you will be alerted, and should adjust the existing position's size by hand."
          }
          confirmLabel="Arm anyway"
          onCancel={() => setDupWarning(null)}
          onConfirm={() => { setDupWarning(null); void handleArm(true); }}
        />
      )}
    </div>
  );
}

interface BrokerPickerProps {

export function BrokerPicker({ selected, onChange }: BrokerPickerProps) {
  const { ibkrPaper, tradierPaper, tastytradePaper } = useSettingsStore();
  const ibkrOk    = !!ibkrPaper.accountId;
  const tradierOk = !!tradierPaper.token && !!tradierPaper.accountId;
  const tastyOk   = !!tastytradePaper.clientSecret && !!tastytradePaper.refreshToken && !!tastytradePaper.accountId;

  return (
    <div className={styles.wrap}>
      <Text size={11} tone="dim">Execute via</Text>
      <div className={styles.row}>
        {([
          { key: 'ibkr_paper' as OptionsBroker, label: 'IBKR Paper', ok: ibkrOk },
          { key: 'tradier_paper' as OptionsBroker, label: 'Tradier Paper', ok: tradierOk },
          { key: 'tastytrade_paper' as OptionsBroker, label: 'Tastytrade Paper', ok: tastyOk },
        ] as const).map(({ key, label, ok }) => (
          <button
            key={key}
            type="button"
            className={`${styles.btn} ${selected === key ? styles.btnActive : ''}`}
            disabled={!ok}
            onClick={() => onChange(key)}
            title={!ok ? 'Configure in Settings → Brokers' : undefined}
          >
            <Text size={12} weight={selected === key ? 600 : 400}>{label}</Text>
            {!ok && <Text size={10} tone="dim">Not configured</Text>}
          </button>
        ))}
      </div>
    </div>
  );
}
